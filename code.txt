function view(v) {
  const allowedViews = [
    "home",
    "discover",
    "groups",
    "activity",
    "profile",
    "detail",
    "create"
  ];

  if (!allowedViews.includes(v)) {
    console.error("Kafo: Unknown page:", v);
    return;
  }

  if (v === "create") {
    showCreate();
    return;
  }

  const target = document.getElementById("view-" + v);

  if (!target) {
    console.error(
      `Kafo: The page section #view-${v} does not exist in index.html.`
    );

    if (typeof toast === "function") {
      toast("This page could not be opened. Please check the page setup.");
    }

    return;
  }

  // Display exactly one page.
  document.querySelectorAll(".view").forEach(section => {
    section.classList.toggle("active", section === target);
  });

  // Update the active navigation item.
  document.querySelectorAll("[data-view]").forEach(button => {
    const isActive = button.dataset.view === v;

    button.classList.toggle("active", isActive);

    if (isActive) {
      button.setAttribute("aria-current", "page");
    } else {
      button.removeAttribute("aria-current");
    }
  });

  // Update the breadcrumb.
  const breadcrumb = document.getElementById("crumb");

  if (breadcrumb) {
    const titles = {
      home: "Home",
      discover: "Discover",
      groups: "My Groups",
      activity: "Activity",
      profile: "Profile",
      detail: "Group Details"
    };

    breadcrumb.textContent = titles[v] || v;
  }

  // Close the mobile sidebar after navigation.
  const sidebar = document.getElementById("sidebar");

  if (sidebar) {
    sidebar.classList.remove("open");
  }

  // Remember the current page in the URL.
  history.replaceState(null, "", "#" + v);

  // Render the selected page.
  try {
    switch (v) {
      case "home":
        if (typeof home === "function") home();
        break;

      case "discover":
        if (typeof renderDiscover === "function") renderDiscover();
        break;

      case "groups":
        if (typeof renderGroups === "function") renderGroups();
        break;

      case "activity":
        if (typeof renderActivity === "function") renderActivity();
        break;

      case "profile":
        if (typeof profileInit === "function") profileInit();
        if (typeof renderAccountSummary === "function") {
          renderAccountSummary();
        }
        break;

      case "detail":
        if (typeof renderDetail === "function") renderDetail();
        break;
    }
  } catch (error) {
    console.error("Kafo: Page rendering failed:", error);

    if (typeof toast === "function") {
      toast("Something went wrong while opening this page.");
    }
  }

  window.scrollTo({
    top: 0,
    behavior: "auto"
  });
}

function profileUpdate(name, email, phone = "+220 700 0000", language = "English") {
  const safeName = String(name || "").trim() || "Kafo Member";
  const safeEmail = String(email || "").trim();
  const safePhone = String(phone || "").trim() || "+220 700 0000";
  const safeLanguage = String(language || "English");

  const initials = safeName
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, 2)
    .map(part => part.charAt(0))
    .join("")
    .toUpperCase();

  const heading = document.querySelector("#profileHeading");
  const nameField = document.querySelector("#profileName");
  const emailField = document.querySelector("#profileEmail");
  const phoneFields = document.querySelectorAll("#profilePhone, #profilePhoneCard");
  const languageFields = document.querySelectorAll("#profileLanguage");

  if (heading) heading.textContent = safeName;
  if (nameField) nameField.textContent = safeName;
  if (emailField) emailField.textContent = safeEmail || "Not added";
  phoneFields.forEach(field => {
  field.textContent = safePhone;
});

languageFields.forEach(field => {
  field.textContent = safeLanguage;
});

  document.querySelectorAll(".user-row b").forEach(element => {
    element.textContent = safeName;
  });

  document.querySelectorAll(".avatar").forEach(element => {
    element.textContent = initials || "KM";
  });

  const topUser = document.querySelector(".user-top");
  if (topUser) {
    const nameSpan = topUser.querySelector("span:not(.avatar)");
    if (nameSpan) nameSpan.textContent = safeName;
  }

  try {
    localStorage.setItem(
      "kafoProfile",
      JSON.stringify({
        name: safeName,
        email: safeEmail,
        phone: safePhone,
        language: safeLanguage
      })
    );
  } catch (error) {
    console.warn("Kafo could not save the profile in this browser.", error);
  }

  renderAccountSummary();
}

document.addEventListener("click", function (event) {
  const target = event.target;

  if (!(target instanceof Element)) return;

  // Main navigation: Home, Discover, My Groups, Activity, Profile.
  const navButton = target.closest("[data-view]");

  if (navButton) {
    event.preventDefault();
    view(navButton.dataset.view);
    return;
  }

  // Actions such as creating a group or sharing a group.
  const actionButton = target.closest("[data-action]");

  if (actionButton) {
    event.preventDefault();

    if (typeof action === "function") {
      action(actionButton.dataset.action);
    }

    return;
  }

  // Open a group card or group row.
  const groupItem = target.closest("[data-open]");

  if (groupItem) {
    event.preventDefault();

    const groupId = groupItem.dataset.open;

    if (groupId) {
      openGroup(groupId);
    }

    return;
  }

  // Open a contribution, expense, or support dialog.
  const modalButton = target.closest("[data-modal]");

  if (modalButton) {
    event.preventDefault();

    if (typeof showModal === "function") {
      showModal(modalButton.dataset.modal);
    }

    return;
  }

  // Switch between group detail tabs.
  const detailTabButton = target.closest("[data-tab]");

  if (detailTabButton) {
    event.preventDefault();

    detailTab = detailTabButton.dataset.tab;
    renderDetail();
    return;
  }

  // Create-group wizard: next step.
  const nextButton = target.closest("[data-next]");

  if (nextButton) {
    const nextStep = Number(nextButton.dataset.next);
    const purpose = document.querySelector(
      '#createForm [name="purpose"]:checked'
    );
    const nameInput = document.getElementById("newName");

    if (nextStep === 2 && !purpose) {
      toast("Please choose a purpose first.");
      return;
    }

    if (nextStep === 3 && (!nameInput || !nameInput.value.trim())) {
      step(2);

      if (nameInput) nameInput.focus();

      toast("Please enter a group name.");
      return;
    }

    step(nextStep);
    return;
  }

  // Create-group wizard: previous step.
  const previousButton = target.closest("[data-prev]");

  if (previousButton) {
    step(Number(previousButton.dataset.prev));
    return;
  }

  // Discover cause filters.
  const categoryButton = target.closest("[data-cat]");

  if (categoryButton) {
    category = categoryButton.dataset.cat;

    document.querySelectorAll("[data-cat]").forEach(button => {
      const active = button === categoryButton;

      button.classList.toggle("active", active);
      button.setAttribute("aria-pressed", String(active));
    });

    renderDiscover();
  }
});
